🚀 USAGE EXAMPLES
================================================================================

1. Basic Benchmarking (Phase 1)
───────────────────────────────
from pgmpy.benchmark import BenchmarkRunner, ErdosRenyiSimulator
from pgmpy.estimators import PC

runner = BenchmarkRunner(
    simulators=[ErdosRenyiSimulator(n_nodes=10, edge_prob=0.3, n_samples=500)],
    methods=[PC(ci_test='pearsonr')],
    metrics=['shd', 'precision_recall'],
    n_runs=20,
)
results = runner.run()
results.summary()  # Print summary statistics
results.to_json()  # Export JSON

3. Chain-of-Thought Tracing (Phase 3)
─────────────────────────────────────
from pgmpy.benchmark import ChainOfThoughtTracer, Explanation

tracer = ChainOfThoughtTracer(method_name="PC", simulator_name="ErdosRenyi")
tracer.add_step("run_algorithm", "Estimating DAG...")
tracer.add_step("compute_metric", "SHD = 3", {"metric": "shd"})
print(tracer.get_trace_string())  # Human-readable trace
print(tracer.to_json())  # Serializable format

explanation = Explanation(
    method_name="PC",
    composite_score=0.82,
    semantic_context={"domain": "biological"},
    fired_rules=["biological_network_orientation"],
    component_scores={"shd": 0.8, "f1": 0.82},
)
print(explanation.to_narrative())
# → "Method PC achieved good performance..."
